  ```ts
  export type StorageLike = {
    createSignedUrl(path: string, expiresIn: number): Promise<{ data: { signedUrl: string } | null }>;
  };

  /** Подписать объект и спросить сканер. null — чисто, строка — вернуть reject(), undefined — сервис недоступен (503). */
  export async function checkStoredFile(
    storage: StorageLike,
    path: string,
    kind: ScanKind,
    onInfected?: () => void
  ): Promise<{ problem: string } | { unavailable: string } | { ok: true }> {
    const { data: signed } = await storage.createSignedUrl(path, 180);
    if (!signed?.signedUrl) return { unavailable: "upload unavailable" };
    const scan = await scanWithService(signed.signedUrl, kind, path);
    if (scan.verdict === "rejected" || scan.verdict === "infected") {
      if (scan.verdict === "infected") onInfected?.();
      return { problem: scan.reason };
    }
    if (scan.verdict === "error") return { unavailable: scan.reason };
    return { ok: true };
  }
  ```

## Термины и технологии

| Термин | Объяснение |
|---|---|
| Zip-бомба (zip bomb) | Архив, который сам весит немного, но при распаковке превращается в терабайты — рассчитан на то, чтобы положить сервер по памяти или диску. `map-scan.ts` защищается не доверием к заголовку (его пишет автор архива), а общим «бюджетом» в 512 МБ на распакованные байты: распаковка идёт по-настоящему, и как только лимит превышен, поток обрывается, не успев занять всю память. |
| Магические байты (file signature / magic bytes) | Первые несколько байт файла, которые жёстко задаёт формат (например, `MZ` у Windows-программы, `PK\x03\x04` у zip). Расширение файла можно переименовать, первые байты — нет без порчи самого файла. `sniffForbidden()` в `map-scan.ts` смотрит именно на эти байты у каждой записи внутри архива, а не на её имя — это и ловит exe, переименованный в `world.dat`. |
| Fail-closed vs fail-open (отказ закрыт / отказ открыт) | Два способа реагировать на сбой проверки: «открыт» пропускает, если проверка не смогла ответить (дешевле для пользователя, опаснее для системы), «закрыт» — наоборот. `scanner-client.ts` намеренно разводит их по времени: пока сервиса нет совсем (`SCANNER_URL` не задан), это осознанный fail-open («skipped»); как только адрес появится, любой сбой сервиса — это отказ (fail-closed), а не тихий пропуск. |
| Дискриминированное объединение (discriminated union) | Приём TypeScript, когда тип — это не один объект с кучей необязательных полей, а список вариантов, различаемых одним общим полем («тегом»). `ScanOutcome` в `scanner-client.ts` — ровно такой тип: `verdict` называет, какой из пяти вариантов перед тобой (`skipped`/`clean`/`rejected`/`infected`/`error`), и у каждого варианта только те поля, которые ему нужны (`reason` есть у отказа, но не у «чисто»). |
| Потоковая распаковка с ограничением (streaming decompression with a cap) | Вместо того чтобы распаковать файл целиком и потом измерить результат (а архив уже мог занять всю память), поток читается кусками, и счётчик уменьшается на каждом кусочке. `drain()` в `map-scan.ts` обрывает распаковщик ровно в момент превышения лимита — архив-бомба не успевает создать большой объект в памяти вообще. |
| EOCD / центральный каталог zip | Формат zip хранит список файлов (кто где лежит, какого размера) в «центральном каталоге» в конце архива, а не перед каждым файлом. `End Of Central Directory` (EOCD) — запись, которая говорит, где этот каталог начинается. `map-scan.ts` сверяет каталог с настоящими локальными заголовками файлов — потому что разные программы читают zip по-разному (одни по каталогу, другие потоком сначала), и архив, где это две разные истории, обманул бы одну из них. |
| Пересборка изображения (image re-encoding) | Вместо того чтобы искать вредоносный «хвост» за честным заголовком PNG/JPEG, картинка полностью раскрывается в пиксели и собирается заново с нуля через библиотеку `sharp`. Всё, что не было пикселями (скрытый код, EXIF с геолокацией телефона), просто не переживает этот шаг — не нужно знать все возможные угрозы, достаточно того, что результат гарантированно состоит только из картинки. |
| Двухфазная загрузка (`pending-` → `confirm`) | Шлюз загрузки сначала выдаёт токен только на временный путь с префиксом `pending-`, и лишь отдельный шаг `confirm` — после всех проверок — переносит файл на постоянное место. Главное свойство: токен не называет финальный адрес, поэтому шаг `confirm` нельзя обойти, загрузив файл и сразу вписав временную ссылку в карту или профиль — на `pending-`-объект нигде не разрешено ссылаться. |
| `AbortSignal.timeout()` | Встроенный в современный `fetch` способ отменить сетевой запрос, если он идёт слишком долго, без ручного `setTimeout` + `AbortController`. `scanner-client.ts` ставит таймаут 150 секунд — чуть меньше, чем лимит самой функции Vercel (300 с), чтобы вовремя получить понятную ошибку, а не оборваться по таймауту платформы молча. |
| `npm audit --omit=dev` против полного `npm audit` | Первая команда проверяет только зависимости, которые реально попадают на сайт (`dependencies` в `package.json`); вторая — ещё и инструменты разработки (`eslint`, `vitest` и то, что они тянут за собой). Путать их опасно: находка в `eslint` не угрожает ни одному посетителю сайта, а находка в `next` угрожает каждому. Этот период обе цифры наконец совпадают: закрыта критическая находка в продакшен-зависимости (`next`), и 0 там держится отдельно от того, что происходит в dev-инструментах. |
| `CHECK`-ограничение в Postgres | Правило на таблице, которое база проверяет сама при каждой записи и которое код обойти не может, даже ошибившись. Миграция `20261101120000` добавляет `check (avatar_url !~ '/pending-')` на `profiles` — значит, даже если какой-то путь в коде забудет проверить, что адрес не временный, сама база откажет записать такую строку. |
